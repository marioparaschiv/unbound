cmake_minimum_required(VERSION 3.22.1)
project(astral_android CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(ASTRAL_REPOSITORY_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../../../../)
add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/libffi ${CMAKE_BINARY_DIR}/libffi)
if(ASTRAL_BUNDLE_URL)
    message(STATUS "Default bundle URL: ${ASTRAL_BUNDLE_URL}")
    add_compile_definitions(ASTRAL_DEFAULT_BUNDLE_URL="${ASTRAL_BUNDLE_URL}")
endif()
set(ASTRAL_CORE_FFI_TARGET astral_ffi CACHE STRING "" FORCE)
add_subdirectory(${ASTRAL_REPOSITORY_ROOT}/core ${CMAKE_BINARY_DIR}/core)

file(GLOB ASTRAL_ANDROID_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/*.cpp)
add_library(astral SHARED ${ASTRAL_ANDROID_SOURCES})
target_compile_options(astral PRIVATE -fvisibility=hidden -fexceptions -frtti)
target_link_libraries(astral PRIVATE astral_core log android)
target_link_options(astral PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL)
