cmake_minimum_required(VERSION 3.22)
project(astral_core C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(ASTRAL_VENDOR_DIR ${CMAKE_CURRENT_LIST_DIR}/../vendor)

option(ASTRAL_CORE_BUNDLE_JSI "Compile jsi.cpp into the core library" ON)
option(ASTRAL_CORE_TESTS "Build the core host tests" OFF)
option(ASTRAL_CORE_FUZZ "Build the native bridge libFuzzer target" OFF)
option(ASTRAL_CORE_JSI_FUZZ "Build the JSI conversion fuzzer against a host Hermes" OFF)
set(ASTRAL_HERMES_SOURCE_DIR "" CACHE PATH "Hermes checkout whose jsi.h matches vendor/react/jsi/jsi.h")

if(ASTRAL_CORE_JSI_FUZZ)
    if(NOT ASTRAL_CORE_TESTS OR NOT ASTRAL_HERMES_SOURCE_DIR)
        message(FATAL_ERROR "ASTRAL_CORE_JSI_FUZZ needs ASTRAL_CORE_TESTS=ON and ASTRAL_HERMES_SOURCE_DIR")
    endif()
    set(ASTRAL_CORE_BUNDLE_JSI OFF CACHE BOOL "" FORCE)
    set(HERMES_ENABLE_TEST_SUITE OFF CACHE BOOL "" FORCE)
    set(HERMES_ENABLE_INTL OFF CACHE BOOL "" FORCE)
    set(HERMES_BUILD_APPLE_FRAMEWORK OFF CACHE BOOL "" FORCE)
    add_subdirectory(${ASTRAL_HERMES_SOURCE_DIR} hermes EXCLUDE_FROM_ALL)
endif()
set(ASTRAL_CORE_FFI_TARGET "" CACHE STRING "libffi target that enables ffi.symbol and ffi.call")

if(ASTRAL_CORE_TESTS AND NOT ASTRAL_CORE_FFI_TARGET)
    find_path(ASTRAL_HOST_FFI_INCLUDE_DIR ffi.h
        HINTS /opt/homebrew/opt/libffi/include /usr/local/opt/libffi/include
        PATH_SUFFIXES ffi)
    find_library(ASTRAL_HOST_FFI_LIBRARY ffi
        HINTS /opt/homebrew/opt/libffi/lib /usr/local/opt/libffi/lib)
    if(ASTRAL_HOST_FFI_INCLUDE_DIR AND ASTRAL_HOST_FFI_LIBRARY)
        add_library(astral_host_ffi INTERFACE)
        target_include_directories(astral_host_ffi INTERFACE ${ASTRAL_HOST_FFI_INCLUDE_DIR})
        target_link_libraries(astral_host_ffi INTERFACE ${ASTRAL_HOST_FFI_LIBRARY})
        set(ASTRAL_CORE_FFI_TARGET astral_host_ffi)
    endif()
endif()

set(ASTRAL_BUNDLE_SIGNING_KEY_FILE ${CMAKE_CURRENT_LIST_DIR}/keys/bundle-signing.pub CACHE FILEPATH "Base64 Ed25519 public key that client bundles must be signed with")
set(ASTRAL_BUNDLE_SIGNING_KEY "")
if(EXISTS ${ASTRAL_BUNDLE_SIGNING_KEY_FILE})
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ASTRAL_BUNDLE_SIGNING_KEY_FILE})
    file(READ ${ASTRAL_BUNDLE_SIGNING_KEY_FILE} ASTRAL_BUNDLE_SIGNING_KEY)
    string(STRIP "${ASTRAL_BUNDLE_SIGNING_KEY}" ASTRAL_BUNDLE_SIGNING_KEY)
endif()
if(ASTRAL_BUNDLE_SIGNING_KEY)
    message(STATUS "Bundle signature verification: on, key from ${ASTRAL_BUNDLE_SIGNING_KEY_FILE}")
else()
    message(STATUS "Bundle signature verification: off, ${ASTRAL_BUNDLE_SIGNING_KEY_FILE} does not exist")
endif()

set(ASTRAL_PR_NUMBER "$ENV{ASTRAL_PR_NUMBER}" CACHE STRING "Pull request number a test build trusts the PR bundle key for; empty for production builds")
set(ASTRAL_PR_BUNDLE_SIGNING_KEY_FILE ${CMAKE_CURRENT_LIST_DIR}/keys/pr-bundle-signing.pub CACHE FILEPATH "Base64 Ed25519 public key that pull request test bundles may be signed with")
set(ASTRAL_PR_BUNDLE_SIGNING_KEY "")
string(LENGTH "${ASTRAL_PR_NUMBER}" ASTRAL_PR_NUMBER_LENGTH)
if(ASTRAL_PR_NUMBER STREQUAL "")
    set(ASTRAL_PR_NUMBER 0)
elseif(NOT ASTRAL_PR_NUMBER MATCHES "^[1-9][0-9]*$" OR ASTRAL_PR_NUMBER_LENGTH GREATER 9)
    message(FATAL_ERROR "ASTRAL_PR_NUMBER must be a pull request number, not \"${ASTRAL_PR_NUMBER}\"")
else()
    if(NOT EXISTS ${ASTRAL_PR_BUNDLE_SIGNING_KEY_FILE})
        message(FATAL_ERROR "A pull request build needs ${ASTRAL_PR_BUNDLE_SIGNING_KEY_FILE}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ASTRAL_PR_BUNDLE_SIGNING_KEY_FILE})
    file(READ ${ASTRAL_PR_BUNDLE_SIGNING_KEY_FILE} ASTRAL_PR_BUNDLE_SIGNING_KEY)
    string(STRIP "${ASTRAL_PR_BUNDLE_SIGNING_KEY}" ASTRAL_PR_BUNDLE_SIGNING_KEY)
    message(STATUS "Pull request build #${ASTRAL_PR_NUMBER}: also trusts ${ASTRAL_PR_BUNDLE_SIGNING_KEY_FILE}")
endif()

file(GLOB_RECURSE ASTRAL_BEARSSL_SOURCES CONFIGURE_DEPENDS ${ASTRAL_VENDOR_DIR}/bearssl/src/*.c)
add_library(astral_bearssl STATIC ${ASTRAL_BEARSSL_SOURCES})
target_include_directories(astral_bearssl PRIVATE ${ASTRAL_VENDOR_DIR}/bearssl/src)
target_include_directories(astral_bearssl PUBLIC ${ASTRAL_VENDOR_DIR}/bearssl/inc)
target_compile_options(astral_bearssl PRIVATE -fvisibility=hidden)

file(GLOB ASTRAL_CORE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/src/*.cpp)
list(APPEND ASTRAL_CORE_SOURCES
    ${ASTRAL_VENDOR_DIR}/monocypher/monocypher.c
    ${ASTRAL_VENDOR_DIR}/monocypher/monocypher-ed25519.c)
if(ASTRAL_CORE_BUNDLE_JSI)
    list(APPEND ASTRAL_CORE_SOURCES ${ASTRAL_VENDOR_DIR}/react/jsi/jsi.cpp)
endif()

if(ASTRAL_CORE_FFI_TARGET)
    file(GLOB ASTRAL_CORE_FFI_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/src/ffi/*.cpp)
    list(APPEND ASTRAL_CORE_SOURCES ${ASTRAL_CORE_FFI_SOURCES})
endif()

add_library(astral_core STATIC ${ASTRAL_CORE_SOURCES})
target_link_libraries(astral_core PRIVATE astral_bearssl)
if(ASTRAL_CORE_FFI_TARGET)
    target_link_libraries(astral_core PUBLIC ${ASTRAL_CORE_FFI_TARGET})
endif()
if(ASTRAL_CORE_JSI_FUZZ)
    target_link_libraries(astral_core PUBLIC jsi)
endif()
target_include_directories(astral_core PUBLIC
    ${CMAKE_CURRENT_LIST_DIR}/include
    ${ASTRAL_VENDOR_DIR}
    ${ASTRAL_VENDOR_DIR}/react)
target_compile_options(astral_core PRIVATE -fvisibility=hidden $<$<COMPILE_LANGUAGE:CXX>:-fexceptions -frtti>)

if(NOT ASTRAL_VERSION)
    file(STRINGS ${CMAKE_CURRENT_LIST_DIR}/../ios/control ASTRAL_CONTROL_VERSION REGEX "^Version:")
    string(REGEX REPLACE "^Version:[ \t]*" "" ASTRAL_VERSION "${ASTRAL_CONTROL_VERSION}")
endif()
target_compile_definitions(astral_core PRIVATE
    ASTRAL_VERSION="${ASTRAL_VERSION}"
    ASTRAL_BUNDLE_SIGNING_KEY="${ASTRAL_BUNDLE_SIGNING_KEY}"
    ASTRAL_PR_NUMBER=${ASTRAL_PR_NUMBER}
    ASTRAL_PR_BUNDLE_SIGNING_KEY="${ASTRAL_PR_BUNDLE_SIGNING_KEY}")

if(ASTRAL_CORE_TESTS)
    enable_testing()
    file(GLOB ASTRAL_CORE_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp)
    if(NOT ASTRAL_CORE_FFI_TARGET)
        list(FILTER ASTRAL_CORE_TEST_SOURCES EXCLUDE REGEX "native_ffi_call[^/]*\\.cpp$")
    endif()
    if(NOT APPLE)
        list(FILTER ASTRAL_CORE_TEST_SOURCES EXCLUDE REGEX "loader_crypto_commoncrypto\\.cpp$")
    endif()
    foreach(test_source ${ASTRAL_CORE_TEST_SOURCES})
        get_filename_component(test_name ${test_source} NAME_WE)
        add_executable(${test_name} ${test_source})
        target_link_libraries(${test_name} PRIVATE astral_core)
        target_compile_definitions(${test_name} PRIVATE ASTRAL_BOOTSTRAP_DIR="${CMAKE_CURRENT_LIST_DIR}/../bootstrap" ASTRAL_CORE_TEST_VECTORS_DIR="${CMAKE_CURRENT_LIST_DIR}/tests/vectors")
        add_test(NAME ${test_name} COMMAND ${test_name})
    endforeach()
endif()

if(ASTRAL_CORE_FUZZ)
    add_executable(native_bridge_libfuzzer ${CMAKE_CURRENT_LIST_DIR}/tests/fuzz/native_bridge_libfuzzer.cpp)
    target_link_libraries(native_bridge_libfuzzer PRIVATE astral_core)
    target_compile_options(native_bridge_libfuzzer PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(native_bridge_libfuzzer PRIVATE -fsanitize=fuzzer,address,undefined)
endif()

if(ASTRAL_CORE_JSI_FUZZ)
    foreach(jsi_fuzz_target native_jsi_fuzz native_jsi_libfuzzer)
        if(jsi_fuzz_target STREQUAL native_jsi_libfuzzer AND NOT ASTRAL_CORE_FUZZ)
            continue()
        endif()
        add_executable(${jsi_fuzz_target} ${CMAKE_CURRENT_LIST_DIR}/tests/jsi-fuzz/${jsi_fuzz_target}.cpp)
        target_link_libraries(${jsi_fuzz_target} PRIVATE astral_core hermesvm_a)
        target_include_directories(${jsi_fuzz_target} PRIVATE ${ASTRAL_HERMES_SOURCE_DIR}/API ${ASTRAL_HERMES_SOURCE_DIR}/public)
    endforeach()
    if(ASTRAL_CORE_FUZZ)
        target_compile_options(native_jsi_libfuzzer PRIVATE -fsanitize=fuzzer,address,undefined)
        target_link_options(native_jsi_libfuzzer PRIVATE -fsanitize=fuzzer,address,undefined)
    endif()
    add_test(NAME native_jsi_fuzz COMMAND native_jsi_fuzz)
    add_executable(native_platform_hermes ${CMAKE_CURRENT_LIST_DIR}/tests/hermes/native_platform_hermes.cpp)
    target_link_libraries(native_platform_hermes PRIVATE astral_core hermesvm_a)
    target_include_directories(native_platform_hermes PRIVATE ${ASTRAL_HERMES_SOURCE_DIR}/API ${ASTRAL_HERMES_SOURCE_DIR}/public)
    add_test(NAME native_platform_hermes COMMAND native_platform_hermes)
    add_executable(native_preload_hermes ${CMAKE_CURRENT_LIST_DIR}/tests/hermes/native_preload_hermes.cpp)
    target_link_libraries(native_preload_hermes PRIVATE astral_core hermesvm_a)
    target_include_directories(native_preload_hermes PRIVATE ${ASTRAL_HERMES_SOURCE_DIR}/API ${ASTRAL_HERMES_SOURCE_DIR}/public)
    target_compile_definitions(native_preload_hermes PRIVATE ASTRAL_BOOTSTRAP_DIR="${CMAKE_CURRENT_LIST_DIR}/../bootstrap" ASTRAL_HERMESC="$<TARGET_FILE:hermesc>")
    add_dependencies(native_preload_hermes hermesc)
    add_test(NAME native_preload_hermes COMMAND native_preload_hermes)
    add_executable(native_fabric_hermes ${CMAKE_CURRENT_LIST_DIR}/tests/hermes/native_fabric_hermes.cpp)
    target_link_libraries(native_fabric_hermes PRIVATE astral_core hermesvm_a)
    target_include_directories(native_fabric_hermes PRIVATE ${ASTRAL_HERMES_SOURCE_DIR}/API ${ASTRAL_HERMES_SOURCE_DIR}/public)
    add_test(NAME native_fabric_hermes COMMAND native_fabric_hermes)
    add_executable(native_handle_leases_hermes ${CMAKE_CURRENT_LIST_DIR}/tests/hermes/native_handle_leases_hermes.cpp)
    target_link_libraries(native_handle_leases_hermes PRIVATE astral_core hermesvm_a)
    target_include_directories(native_handle_leases_hermes PRIVATE ${ASTRAL_HERMES_SOURCE_DIR}/API ${ASTRAL_HERMES_SOURCE_DIR}/public)
    add_test(NAME native_handle_leases_hermes COMMAND native_handle_leases_hermes)
    add_executable(native_struct_layouts_hermes ${CMAKE_CURRENT_LIST_DIR}/tests/hermes/native_struct_layouts_hermes.cpp)
    target_link_libraries(native_struct_layouts_hermes PRIVATE astral_core hermesvm_a)
    target_include_directories(native_struct_layouts_hermes PRIVATE ${ASTRAL_HERMES_SOURCE_DIR}/API ${ASTRAL_HERMES_SOURCE_DIR}/public)
    add_test(NAME native_struct_layouts_hermes COMMAND native_struct_layouts_hermes)
    add_executable(native_runtime_hermes ${CMAKE_CURRENT_LIST_DIR}/tests/hermes/native_runtime_hermes.cpp)
    target_link_libraries(native_runtime_hermes PRIVATE astral_core hermesvm_a)
    target_include_directories(native_runtime_hermes PRIVATE ${ASTRAL_HERMES_SOURCE_DIR}/API ${ASTRAL_HERMES_SOURCE_DIR}/public)
    target_compile_definitions(native_runtime_hermes PRIVATE
        ASTRAL_BOOTSTRAP_DIR="${CMAKE_CURRENT_LIST_DIR}/../bootstrap"
        ASTRAL_TEST_FFI=$<IF:$<BOOL:${ASTRAL_CORE_FFI_TARGET}>,1,0>)
    add_test(NAME native_runtime_hermes COMMAND native_runtime_hermes)
endif()
